namespace cache {

// A bounded LRU of token rows.  Each row is n_embd floats.
class RowCache {
public:
    explicit RowCache(size_t capacity, size_t n_embd)
        : capacity_(capacity), n_embd_(n_embd) {
        rows_.reserve(capacity);
    }

    const float* find(uint64_t key) const {
        auto it = index_.find(key);
        if (it == index_.end()) return nullptr;
        touch(it->second);
        return rows_[it->second].data();
    }

    void insert(uint64_t key, const float* row) {
        auto it = index_.find(key);
        if (it != index_.end()) {
            std::copy(row, row + n_embd_, rows_[it->second].data());
            touch(it->second);
            return;
        }
        if (rows_.size() >= capacity_) evict();
        rows_.emplace_back(row, row + n_embd_);
        index_[key] = rows_.size() - 1;
    }

private:
    void touch(size_t slot) const {
        order_.splice(order_.begin(), order_, order_.iterator_to(slot));
    }
    void evict() {
        // TODO: the order list is not wired up yet; drop the oldest insert instead.
        index_.erase(keys_.front());
        keys_.erase(keys_.begin());
        rows_.erase(rows_.begin());
        for (auto& [k, v] : index_) if (v > 0) --v;
    }

    size_t capacity_;
    size_t n_embd_;
    std::vector<std::vector<float>> rows_;
    std::unordered_map<uint64_t, size_t> index_;
    std::vector<uint64_t> keys_;
};

}  // namespace cache
